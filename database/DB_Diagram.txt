
Table clientes {
  cliente_id int [pk, increment]
  nombre varchar(100) [not null]
  email varchar(150) [unique, not null]
  telefono varchar(20)
  estado varchar(10) [not null, note: 'ACTIVO o INACTIVO']
  limite_credito decimal(12,2) [not null]
}

Table direcciones {
  direccion_id int [pk, increment]
  cliente_id int [not null]
  calle varchar(150) [not null]
  ciudad varchar(100) [not null]
  estado varchar(100) [not null]
  codigo_postal varchar(10) [not null]
}

Table envios {
  envio_id int [pk, increment]
  remitente_id int [not null]
  destinatario_id int [not null]
  direccion_origen_id int [not null]
  direccion_destino_id int [not null]
  costo_total decimal(12,2) [not null]
  fecha DATETIME [not null]
  estado varchar(20) [not null, note: 'REGISTRADO, EN_TRANSITO, ENTREGADO, CANCELADO']
}

Table almacenes {
  almacen_id int [pk, increment]
  nombre varchar(100) [not null]
  ubicacion varchar(200) [not null]
  capacidad_maxima int [not null]
}

Table paquetes {
  paquete_id int [pk, increment]
  envio_id int [not null]
  almacen_id int [not null]
  peso decimal(10,2) [not null]
  largo decimal(10,2) [not null]
  ancho decimal(10,2) [not null]
  alto decimal(10,2) [not null]
  contenido varchar(255)
}

Table historial_rastreo {
  rastreo_id int [pk, increment]
  envio_id int [not null]
  fecha DATETIME [not null]
  ubicacion_actual varchar(200) [not null]
  estado_evento varchar(30) [not null]
  observaciones varchar(255)
}

Table auditoria_envios {
  auditoria_id int [pk, increment]
  envio_id int [not null]
  costo_anterior decimal(12,2) [not null]
  costo_nuevo decimal(12,2) [not null]
  usuario varchar(100) [not null]
  fecha_modificacion DATETIME [not null]
}

Ref: direcciones.cliente_id > clientes.cliente_id
Ref: envios.remitente_id > clientes.cliente_id
Ref: envios.destinatario_id > clientes.cliente_id
Ref: envios.direccion_origen_id > direcciones.direccion_id
Ref: envios.direccion_destino_id > direcciones.direccion_id
Ref: paquetes.envio_id > envios.envio_id
Ref: paquetes.almacen_id > almacenes.almacen_id
Ref: historial_rastreo.envio_id > envios.envio_id
Ref: auditoria_envios.envio_id > envios.envio_id
